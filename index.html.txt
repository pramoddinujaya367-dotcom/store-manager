        function addToCart(id) {
            let item = items.find(i => i.id === id);
            if (!item || item.qty <= 0) {
                alert('මෙම භාණ්ඩය තොගයේ නොමැත!');
                return;
            }
            let cartItem = cart.find(c => c.id === id);
            if (cartItem) {
                if (cartItem.qty < item.qty) cartItem.qty++;
            } else {
                cart.push({ id: item.id, name: item.name, price: item.price, cost: item.cost, qty: 1, maxQty: item.qty });
            }
            renderCart();
        }

        function renderCart() {
            let body = document.getElementById('cartTableBody');
            body.innerHTML = '';
            let total = 0;
            cart.forEach((c, idx) => {
                let sub = c.price * c.qty;
                total += sub;
                body.innerHTML += `
                    <tr>
                        <td><small>${c.name}</small></td>
                        <td>
                            <input type="number" value="${c.qty}" min="1" max="${c.maxQty}" class="form-control form-control-sm" style="width: 60px;" onchange="updateCartQty(${idx}, this.value)">
                        </td>
                        <td><small>${c.price}</small></td>
                        <td><small>${sub.toFixed(2)}</small></td>
                        <td><button class="btn btn-sm text-danger" onclick="removeCart(${idx})">&times;</button></td>
                    </tr>
                `;
            });
            document.getElementById('cartTotal').innerText = total.toFixed(2);
        }

        function updateCartQty(idx, val) {
            cart[idx].qty = parseInt(val) || 1;
            renderCart();
        }

        function removeCart(idx) {
            cart.splice(idx, 1);
            renderCart();
        }

        function clearCart() {
            cart = [];
            renderCart();
        }

        function checkout() {
            if (cart.length === 0) {
                alert('බිල්පත හිස්ය!');
                return;
            }

            let totalAmt = cart.reduce((sum, i) => sum + (i.price * i.qty), 0);
            let totalCost = cart.reduce((sum, i) => sum + (i.cost * i.qty), 0);
            let profit = totalAmt - totalCost;

            // Update item quantities
            cart.forEach(c => {
                let stockItem = items.find(i => i.id === c.id);
                if (stockItem) stockItem.qty -= c.qty;
            });

            let saleRecord = {
                id: 'INV-' + Date.now().toString().substring(6),
                time: new Date().toLocaleString(),
                items: [...cart],
                total: totalAmt,
                profit: profit
            };

            sales.unshift(saleRecord);
            saveState();

            // Print Receipt Modal
            renderThermalReceipt(saleRecord);
            let modal = new bootstrap.Modal(document.getElementById('receiptModal'));
            modal.show();

            cart = [];
            renderAll();
        }

                <div class="d-flex justify-content-between"><small>Invoice: ${sale.id}</small><small>${sale.time}</small></div>

        }

        function importData(e) {
            let file = e.target.files[0];
            if (!file) return;
            let reader = new FileReader();
            reader.onload = function(evt) {
                try {
                    let d = JSON.parse(evt.target.result);
                    if (d.items && d.sales) {
                        items = d.items;
                        sales = d.sales;
                        saveState();
                        alert('Backup සාර්ථකව Restore විය!');
                        renderAll();
                    }
                } catch(err) {
                    alert('වැරදි Backup ෆයිල් එකකි!');
                }
            };
            reader.readAsText(file);
        }
